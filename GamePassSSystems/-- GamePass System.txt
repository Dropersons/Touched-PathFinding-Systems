-- GamePass System (Combines modular code where data is fetched in a single server script. Tools given depending on the button clicked by player which sends the specificID to server for prompting pass)

-- Server 

local replicatedStorage = game:GetService("ReplicatedStorage")
local ServerScriptService = game:GetService("ServerScriptService")
local marketPlaceServices = game:GetService("MarketplaceService")

local remotes = replicatedStorage:WaitForChild("Remotes")
local BuyPass = remotes:WaitForChild("BuyPass")

local GameTest = ServerScriptService:WaitForChild("GameTest")
local GiveTool = require(GameTest:WaitForChild("GiveTool"))
local GamePassSystem = require(GameTest:WaitForChild("GamePassSystem"))
local ShopItemConfig = require(GameTest:WaitForChild("ShopItemConfig"))

local players = game:GetService("Players")

BuyPass.OnServerEvent:Connect(function(player , SpecificID)
	GamePassSystem.Prompt(player , SpecificID)
end)

marketPlaceServices.PromptGamePassPurchaseFinished:Connect(function(player , specificId , wasPurchased)
	if wasPurchased then
		if specificId == 1840172665 then
			GiveTool.GiveSpeedCoil(player)
			
		elseif specificId == 1841078487 then
			GiveTool.JumpCoil(player)
		end
		
	end
end)

players.PlayerAdded:Connect(function(player)
	player.CharacterAdded:Connect(function(character)
		local hasSpeedCoil = GamePassSystem.HasPass(player , 1840172665)
		local hasJumpCoil = GamePassSystem.HasPass(player , 1841078487)

		if hasSpeedCoil then
			GiveTool.GiveSpeedCoil(player)
		end

		if hasJumpCoil then
			GiveTool.JumpCoil(player)
		end
	end)
end)

-- GamePassSystem (Module)

local ReplicatedStorage = game:GetService("ReplicatedStorage")
local remotes = ReplicatedStorage:WaitForChild("Remotes")
local BuyPass = remotes:WaitForChild("BuyPass")

local marketPlaceServices = game:GetService("MarketplaceService")
local GamePassSystem = {}

function GamePassSystem.Prompt(player , specificId)
	marketPlaceServices:PromptGamePassPurchase(player , specificId)
end

function GamePassSystem.HasPass(player , specificId)
	return marketPlaceServices:UserOwnsGamePassAsync(player.UserId , specificId)
end

return GamePassSystem

-- GiveTool (Module)

local replicatedStorage = game:GetService("ReplicatedStorage")
local SpeedCoil = replicatedStorage:WaitForChild("SpeedCoil")
local JumpCoil = replicatedStorage:WaitForChild("JumpCoil")

local GiveTool = {}

function GiveTool.GiveSpeedCoil(player)
	if player.Backpack:FindFirstChild("SpeedCoil") or (player.Character and player.Character:FindFirstChild("SpeedCoil")) then
		return
	end
	local Speedclone = SpeedCoil:Clone()
	print("speed coil")
	Speedclone.Grip = CFrame.new(0,1,1) * CFrame.Angles(0,0,math.rad(90))
	Speedclone.Parent = player.Backpack
end

function GiveTool.JumpCoil(player)
	if player.Backpack:FindFirstChild("JumpCoil") or (player.Character and player.Character:FindFirstChild("JumpCoil")) then
		return
	end
	local Jumpclone = JumpCoil:Clone()
	print("jump coil")
	Jumpclone.Grip = CFrame.new(0,1,1) * CFrame.Angles(0,0,math.rad(90))
	Jumpclone.Parent = player.Backpack
end

return GiveTool
